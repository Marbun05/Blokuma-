export default function DinoStage({ islandId, isRunning, onMissionComplete }) {
 const [animating, setAnimating] = useState(false);

 // Konfigurasi latar belakang berdasarkan pulau
 const islandThemes = {
   1: { bg: 'bg-amber-50', obstacle: 'rumah desa', title: 'Desa Urutan' },
   2: { bg: 'bg-emerald-50', obstacle: 'pohon lebat', title: 'Hutan Perulangan' },
   3: { bg: 'bg-slate-100', obstacle: 'batu gunung', title: 'Gunung Kondisi' },
   4: { bg: 'bg-sky-50', obstacle: 'koin emas', title: 'Kota Variabel' },
   5: { bg: 'bg-indigo-50', obstacle: 'rintangan arcade', title: 'Arena Game' },
   6: { bg: 'bg-zinc-900 text-white', obstacle: 'laser lab', title: 'Laboratorium' },
 };

 const currentTheme = islandThemes[islandId] || islandThemes[1];

 useEffect(() => {
   if (isRunning) {
     setAnimating(true);
     // Simulasi durasi animasi berjalan sebelum misi dinyatakan selesai
     const timer = setTimeout(() => {
       setAnimating(false);
       if (onMissionComplete) onMissionComplete();
     }, 3000);
     return () => clearTimeout(timer);
   }
 }, [isRunning]);

 return (
   <div className={`relative h-64 w-full overflow-hidden rounded-2xl border-4 border-gray-700 ${currentTheme.bg} flex items-end p-6 shadow-inner`}>
     {/* Indikator Skor / Info Misi ala Game Dinosaurus */}
     <div className="absolute top-4 right-6 font-mono text-sm font-bold tracking-wider opacity-70">
       PULAU {islandId} • {currentTheme.title.toUpperCase()}
     </div>

     {/* Latar Belakang Bergerak (Parallax Effect) */}
     <div className={`absolute inset-0 flex items-center justify-around opacity-20 ${animating ? 'animate-pulse' : ''}`}>
       {/* Contoh elemen latar belakang */}
       <span className="text-4xl">☁️</span>
       <span className="text-4xl">🌲</span>
       <span className="text-4xl">☁️</span>
     </div>

     {/* Area Arena Bermain */}
     <div className="relative z-10 flex w-full items-end justify-between border-b-4 border-dashed border-gray-600 pb-2">
       {/* Maskot Kiko */}
       <div className={`text-6xl transition-transform duration-300 ${animating && islandId === 2 ? '-translate-y-20' : 'translate-y-0'}`}>
         🤖
       </div>

       {/* Rintangan yang Mendekat */}
       <div className={`text-4xl transition-all duration-1000 ${animating ? '-translate-x-96 opacity-100' : 'translate-x-0 opacity-80'}`}>
         {islandId === 2 ? '🌲' : '🌵'}
       </div>
     </div>
   </div>
 );
}

Saat pengguna menekan tombol **Mainkan 🎮** pada salah satu kartu di menu Galeri, sistem tidak lagi sekadar merespons di halaman yang sama, melainkan mengarahkan (*routing*) ke **halaman permainan baru khusus** (`/app/gallery/play/[game-id]`):

* **Kartu "Hutan Ajaib Melodi":**
 * Route: `/app/gallery/play/hutan-ajaib-melodi`
 * Tampilan Halaman Baru: Panggung penuh bertema musik interaktif lengkap dengan kontrol instrumen audio dan balok nada.
* **Kartu "Kiko Mencari Baterai Emas":**
 * Route: `/app/gallery/play/kiko-baterai-emas`
 * Tampilan Halaman Baru: Arena permainan petualangan labirin penuh di mana pemain mengontrol Kiko untuk mengumpulkan baterai emas.
* **Kartu "Balap Roket Galaksi":**
 * Route: `/app/gallery/play/balap-roket-galaksi`
 * Tampilan Halaman Baru: Game arcade antariksa dengan roket melaju dan papan skor interaktif.

### B. Fitur Tambahan pada Halaman Permainan Khusus
1. **Header Game:** Menampilkan judul permainan, pembuat karya, kelas, serta tombol "Kembali ke Galeri".
2. **Area Layar Game Utama:** Tampilan permainan luas (*full canvas/stage*) dengan tombol kontrol interaktif.
3. **Panel Apresiasi:** Tombol "Beri Suka ❤️" dan kolom komentar apresiasi dari teman-teman.

## 5. Indikator Keberhasilan Misi (Success Modal)
Setelah animasi di **Kiko Stage** selesai dieksekusi dan kondisi misi pulau terpenuhi:
1. Muncul **Success Modal** tepat di tengah layar dengan latar belakang *overlay/backdrop-blur*.
2. Memuat pesan ucapan selamat, jumlah reward XP yang diperoleh, tombol **"Coba Lagi"**, serta tombol **"Lanjut Misi Berikutnya 🚀"**.
3. Tombol **"Lanjut Misi Berikutnya"** otomatis memperbarui status di database untuk membuka kunci (*unlock*) pulau selanjutnya di menu Petualangan.
