import { CURRICULUM_DATA } from '../../../data/curriculumData';

export default function AdventurePage() {
 const [gradeLevel, setGradeLevel] = useState('3-4'); // Default

   if (!className) return '3-4';
   if (className.includes('1') || className.includes('2')) return '1-2';

Kode Fetching Dinamis di Frontend React (app/adventure/page.jsx)
Di modul Petualangan, kamu tinggal mengambil data dari Supabase berdasarkan angka kelas siswa (student_class):
import React, { useState, useEffect } from 'react';
import { supabase } from '../../../../lib/supabaseClient'; // Sesuaikan lokasi client supabase kamu

export default function StudentAdventurePage() {
 const [worlds, setWorlds] = useState([]);
 const [loading, setLoading] = useState(true);
 const [studentClass, setStudentClass] = useState('Kelas 1 SD');

 const extractGradeNumber = (className) => {
   if (!className) return 1;
   const match = className.match(/\d+/);
   return match ? parseInt(match[0], 10) : 1;
 };

 const fetchCurriculumData = async () => {
   setLoading(true);
   const savedClass = localStorage.getItem('student_class') || 'Kelas 1 SD';
   const gradeNum = extractGradeNumber(savedClass);
   setStudentClass(savedClass);

   // Fetch data Supabase filter berdasarkan min_grade
   const { data, error } = await supabase
     .from('curriculum_worlds')
     .select('*')
     .lte('min_grade', gradeNum) // Ambil materi yang min_grade <= kelas siswa
     .order('min_grade', { ascending: true });

   if (!error && data) {
     setWorlds(data);
   }
   setLoading(false);
 };

 useEffect(() => {
   fetchCurriculumData();

   window.addEventListener('student_profile_updated', fetchCurriculumData);
   return () => window.removeEventListener('student_profile_updated', fetchCurriculumData);
 }, []);

 return (
   <div>
     {/* Render Kartu Worlds dari state `worlds` */}
   </div>
 );
}
